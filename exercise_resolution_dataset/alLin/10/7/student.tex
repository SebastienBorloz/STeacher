\[ = \begin{pmatrix} -3\cdot(-2) - (-3)\cdot1 \\ -((-3)(-3) - (-3)(-1)) \\ (-3) \cdot 2 - (-2)(-1) \end{pmatrix} \]
\[ = \begin{pmatrix} 6 + 3 \\ -(9 - 3) \\ -6 + 2 \end{pmatrix} \]
\[ = \begin{pmatrix} 9 \\ -6 \\ -4 \end{pmatrix} \]
\[ \left\| \begin{pmatrix} 9 \\ -6 \\ -4 \end{pmatrix} \right\| = \sqrt{9^2 + (-6)^2 + (-4)^2} \]
\[ = \sqrt{81 + 36 + 16} \]
\[ = \sqrt{133} \]
