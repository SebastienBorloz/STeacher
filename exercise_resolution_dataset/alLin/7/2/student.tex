\text{(dans }[0,360[\text{)}

\[ \Leftrightarrow 3\alpha = \alpha + 40° + k360° \]

\[ \Leftrightarrow 2\alpha = 40° + k360° \]
\[ \Leftrightarrow \alpha = 20° + k180° \]

\[ \text{Si } k=1 : 20° + 180° = 200° \in [0,360[ \]

\[ \Leftrightarrow 3\alpha = 140° - \alpha + k360° \]

\[ \Leftrightarrow 4\alpha = 140° + k360° \]

\text{(a) } $\cos(x) + 2\sin(x)\tan(x) = 1$
\text{dans }[0,360[

\[ \Leftrightarrow 2\sin(x)\tan(x) = 1 - \cos(x) \]
\[ \Leftrightarrow 4\sin^2(x)\tan^2(x) = 1 - \cos^2(x) \]
\[ \Leftrightarrow 4(1-\cos^2(x))\tan^2(x) = 1 - \cos^2(x) \]
\[ \Leftrightarrow 4 - 4\cos^2(x)\tan^2(x) = 1 - \cos^2(x) \]
\[ \Leftrightarrow \cos^2(x) - 4\cos^2(x)\tan^2(x) + 3 = 0 \]
\[ \Leftrightarrow \cos^2(x)(1 - 4\tan^2(x)) + 3 = 0 \]
\[ \Leftrightarrow (-\cos^2(x))(-1 + 4\tan^2(x)) + 3 = 0 \]
\[ \Leftrightarrow (-\cos^2(x))\left(\frac{-1}{4\cos^2(x)}\right) + 3 = 0 \]
