\text{7. a) } \vec{a} = \begin{pmatrix} -3 \\ -2 \\ -3 \end{pmatrix}, \quad \vec{b} = \begin{pmatrix} -1 \\ 1 \\ -3 \end{pmatrix}
\]
\[
\text{Aire: } |\vec{a} \times \vec{b}| = \left| \begin{pmatrix} -3 \\ -2 \\ -3 \end{pmatrix} \times \begin{pmatrix} -1 \\ 1 \\ -3 \end{pmatrix} \right| = \left| \begin{pmatrix} 9 \\ -6 \\ -5 \end{pmatrix} \right| = \sqrt{81 + 36 + 25} = \sqrt{142}
\]
\[
\text{b) hauteur } \rightarrow \vec{c} = \begin{pmatrix} -1 \\ -2 \\ -7 \end{pmatrix}
\]
\[
\text{longueur hauteur: } \left| \begin{pmatrix} -1 \\ -2 \\ -7 \end{pmatrix} \right| = \sqrt{1 + 4 + 49} = \sqrt{54}
\]
\[
\text{Aire base } \cdot \text{ hauteur } = \sqrt{142} \cdot \sqrt{54} = 142 \cdot 54 \quad \text{(volume)}
