\[
\text{8. Si le résultat de } x, y \text{ et } z \text{ est uniquement } 0 \text{ alors c'est linéairement indépendant!}
\]

\[
\begin{cases}
x + 2y + az = 0 \\
x + 2y + 3z = 0 \\
x + 2y - 7z = 0
\end{cases}
\]

\[
\begin{cases}
x + 2y + az = 0 \\
(3-a)z = 0 \\
az + 3z = 0
\end{cases}
\]

\[
\begin{cases}
x + 2y + az = 0 \\
(3-a)z = 0 \quad \text{1} \\
(7+a)z = 0
\end{cases}
\]

\[
\text{Oui ils sont linéairement indépendant car si on remplace} x, y \text{ et } z \text{ par } 0 \text{ alors leségalité fonctionne cela ne dépend pas de } a.
\]
