\[
c=(a,0,1) \quad \vec{d} = \begin{pmatrix} -2 \\ 1 \\ 0 \end{pmatrix}
\]

\[
\vec{CP} \perp \vec{d} \quad \text{où P} \in \vec{d}
\]

\[
\vec{CP} \cdot \vec{d} = 0
\]

\[
\vec{CP} = \vec{OP} - \vec{OC} = \begin{pmatrix} x \\ y \\ z \end{pmatrix} - \begin{pmatrix} a \\ 0 \\ 1 \end{pmatrix} = \begin{pmatrix} x-a \\ y \\ z-1 \end{pmatrix}
\]

\[
\begin{pmatrix} x-a \\ y \\ z-1 \end{pmatrix} \cdot \begin{pmatrix} -2 \\ 1 \\ 0 \end{pmatrix} = 0
\]

\[
(x-a)(-2) + y \cdot 1 + (z-1) \cdot 0 = 0
\]

\[
\Rightarrow -2x + 2a + y + 0 = 0
\]
