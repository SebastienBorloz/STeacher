\[ \text{exercice 9} \]
\[ \begin{cases} x + y + 2z - w = 10 \\ x + y + z + w = 5 \\ x + y + 3z - 3w = 15 \end{cases}\]
\[ \text{3 variables pour 3 équations} \rightarrow \text{système sous-déterminé} \rightarrow \text{infinité de sol!} \]
\[\rightarrow \begin{pmatrix} 1 & 1 & 2 & -1 & | & 10 \\ 1 & 1 & 1 & 1 & | & 5 \\ 1 & 1 & 3 & -3 & | & 15 \end{pmatrix} \]
\[ \begin{pmatrix} 1 & 1 & 2 & -1 & | & 10 \\ 1 & 1 & 1 & 1 & | & 5 \\ 1 & 1 & 3 & -3 & | & 15 \end{pmatrix} \xrightarrow{L_2 \leftarrow L_2 - L_1} \begin{pmatrix} 1 & 1 & 2 & -1 & | & 10 \\ 0 & 0 & -1 & 2 & | & 5 \\ 1 & 1 & 3 & -3 & | & 15 \end{pmatrix} \]
\[ \begin{pmatrix} 1 & 1 & 2 & -1 & | & 10 \\ 0 & 0 & -1 & 2 & | & 5 \\ 1 & 1 & 3 & -3 & | & 15 \end{pmatrix} \xrightarrow{L_3 \leftarrow L_3 - L_1} \begin{pmatrix} 1 & 1 & 2 & -1 & | & 10 \\ 0 & 0 & -1 & 2 & | & 5 \\ 0 & 0 & 1 & -3 & | & 5 \end{pmatrix} \]
\[ \begin{pmatrix} 1 & 1 & 2 & -1 & | & 10 \\ 0 & 0 & -1 & 2 & | & 5 \\ 0 & 0 & 1 & -3 & | & 5 \end{pmatrix} \xrightarrow{L_2 \leftarrow \frac{L_3}{-1}} \begin{pmatrix} 1 & 1 & 2 & -1 & | & 10 \\ 0 & 0 & 1 & -2 & | & -5 \\ 0 & 0 & 1 & -3 & | & 5 \end{pmatrix} \]
