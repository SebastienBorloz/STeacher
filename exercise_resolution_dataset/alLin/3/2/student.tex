\text{exercice 2}
\[ \text{(a) } \cos(x) + 2\sin(x)\tan(x) = 1 \text{ dans } [0, 2\pi[ \]
\[ \Leftrightarrow \cos(x) + 2\sin(x) \frac{\sin(x)}{\cos(x)} = 1 \]
\[ \Leftrightarrow 2\sin(x) \frac{\sin(x)}{\cos(x)} = 1 - \cos(x) \]
\[ \Leftrightarrow 2\sin(x) \cdot \sin(x)  = 1 - \cos^2(x) \]
\[ \Leftrightarrow 2\sin^2(x) = 1 - \cos^2(x) \]
\[ \Leftrightarrow 2(1 - \cos^2(x)) = \cos^2(x) = 1 \]
\[ \Leftrightarrow - 2\cos^2(x) + \cos(x) = -1 \]
\[ \Leftrightarrow -\cos^2(x) = -1 \]
\[ \Leftrightarrow \cos^2(x) = 1 \]
\[ \Leftrightarrow \cos(x) = \sqrt{1} = 1 \]
