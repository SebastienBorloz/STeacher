\text{6 a) } \lim_{x \to \infty} \left( x+1 - \frac{x^2}{x+2} \right) = +\infty+1 - (+\infty) = +\infty+1-\infty = \text{n.d.}
\]

\[
\text{b) } \lim_{x \to 1} \frac{\sqrt{x^2+8} - \sqrt{x^2+15}}{\sqrt{x^2+3} - \sqrt{x^2+24}}
\]
\[
= \frac{\sqrt{1+8} - \sqrt{1+15}}{\sqrt{1+3} - \sqrt{1+24}}
\]
\[
= \frac{\sqrt{9} - \sqrt{16}}{\sqrt{4} - \sqrt{25}}
\]
\[
= \frac{3 - 4}{\sqrt{4} - 5}
\]
\[
= \frac{-1}{\sqrt{9}-5}
\]
\[
= \frac{1}{2\sqrt{2}-5}
\]

\[
\text{c) } \lim_{x \to 1} \frac{\sqrt{x^2+2x+6} - 3}{x^2-1}
\]
\[
= \frac{\sqrt{1+2+6} - 3}{1-1}
\]
\[
= \frac{\sqrt{9} - 3}{0}
\]
\[
= \frac{3 - 3}{0}
\]
\[
= \frac{0}{0} \quad \text{inexact!}
\]

\[
\text{gauche}
\]
\[
\lim_{x \to 1^-} \frac{\sqrt{x^2+2x+6} - 3}{x^2-1}
\]
\[
= \frac{\sqrt{1^-+2^-+6} - 3}{1^--1}
\]
\[
= \frac{\sqrt{9^-} - 3}{0^-}
\]
\[
= \frac{3^- - 3}{0^-}
\]
\[
= \frac{0^-}{0^-} = 0^+
\]

\[
\text{droite}
\]
\[
\lim_{x \to 1^+} \frac{\sqrt{x^2+2x+6} - 3}{x^2-1}
\]
\[
= \frac{\sqrt{1^++2^++6} - 3}{1^+-1}
\]
\[
= \frac{\sqrt{9^+} - 3}{0^+} 
\]
\[
= \frac{3^+ - 3}{0^+}
\]
\[
= \frac{0^+}{0^+} = 0^+
\]

\[
\text{des deux côtés}
\]
\[
\lim_{x \to 1} \frac{\sqrt{x^2+2x+6} - 3}{x^2-1} = 0^+
\]
