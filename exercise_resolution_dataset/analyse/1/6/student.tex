\text{Exercice 6}
\[ \text{a) } \lim_{x \to +\infty} \left( \frac{x(x+2)+1(x+2)-x^2}{x+2} \right) \]
\[ = \lim_{x \to +\infty} \left( \frac{x^2+2x+x+2-x^2}{x+2} \right) \]
\[ = \lim_{x \to +\infty} \left( \frac{3x+2}{x+2} \right) \]
\[ = \lim_{x \to +\infty} \frac{x(3+2/x)}{x(1+2/x)} = 3 \]
\[ \text{b) } \lim_{x \to 1} \frac{\sqrt{x^2+8}-\sqrt{x^2+15}}{\sqrt{x^2+3}-\sqrt{x^2+24}} \]
\[ = \frac{\sqrt{9}-\sqrt{16}}{\sqrt{4}-\sqrt{25}} = \frac{3-4}{2-5} = \frac{-1}{-3} = \frac{1}{3} \]
\[ \text{c) } \lim_{x \to 1} \frac{\sqrt{x^2+2x+6}-3}{x^2-1} \]
\[ = \left[ \frac{\sqrt{1+2+6}-3}{1-1} \right] = \left[ \frac{0}{0} \right] \]
\[ \lim_{x \to 1} \frac{\sqrt{x^2+2x+6}-3}{x^2-1} \cdot \frac{\sqrt{x^2+2x+6}+3}{\sqrt{x^2+2x+6}+3} \]
\[ = \lim_{x \to 1} \frac{x^2+2x+6-3}{(x^2-1)(\sqrt{x^2+2x+6}+3)} \]
\[ = \lim_{x \to 1} \frac{x^2+2x-3}{(x^2-1)(\sqrt{x^2+2x+6}+3)} \]
\[ x^2 + 2x + 3 \implies \Delta = 4 - 12 < 0\]
\[ \frac{1 + 2 + 3}{0 \cdot (\sqrt{9} + 3)} = \frac{6}{0} = \infty\]
\[ = \lim_{x \to 1^+} \frac{x^2 + 2x + 3}{(x^2-1)(\sqrt{x^2+2x+6}+3)} = +\infty \]
\[ = \lim_{x \to 1^-} \frac{x^2 + 2x + 3}{(x^2-1)(\sqrt{x^2+2x+6}+3)} = +\infty \]
