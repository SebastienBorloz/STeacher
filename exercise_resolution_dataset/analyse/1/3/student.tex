\[ \text{Exercice 3} \]
\[ \text{a)} \sum_{k=1}^6 7(10^{-k}) \]
\[ = 7 \sum_{k=1}^6 \frac{1}{10^k} \]
\[ = 7 \sum_{k=1}^6 10^{-k} \]
\[ \implies -k = t\]
\[ = 7 \sum_{t=-1}^t 10^{t} \]
\[ = 7 \left( \frac{10^{5}-1}{10-1} +10^{-1}\right) \]
\[ = 7 \left( \frac{10^{-5} - 1}{9} + \frac{1}{10^1} \right) \]
\[ = 7 \frac{10(10^5-1) + 9}{90} \]
\[ = 7 \frac{10^6 - 10+9}{90} \]
\[ = 7 \frac{10^6 - 1}{90} \]
\[ = \frac{7}{90}(10^6 - 1) \]
