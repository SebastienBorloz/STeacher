\[
\text{a) } \lim_{x \to \infty} \left(x+1 - \frac{x^2}{x+2}\right) = (\infty - \infty)
\]
\[
= \lim_{x \to \infty} \frac{x(x+2) + x+2 - x^2}{x+2}
\]
\[
= \lim_{x \to \infty} \frac{x^2 + 2x + x + 2 - x^2}{x+2}
\]
\[
= \lim_{x \to \infty} \frac{3x+2}{x+2}
\]
\[
= \lim_{x \to \infty} \frac{x(3 + \frac{2}{x})}{x(1 + \frac{2}{x})} = 3
\]

\[
\text{b) } \lim_{x \to 1} \frac{\sqrt{x^2+6} - \sqrt{x^2+15}}{\sqrt{x^2+3} - \sqrt{x^2+24}}
\]
\[
= \frac{\sqrt{9} - \sqrt{16}}{\sqrt{4} - \sqrt{25}}
\]
\[
= \frac{-1}{-3} = \frac{1}{3}
\]

\[
\text{c) } \lim_{x \to 1} \frac{\sqrt{x^2+2x+6} - 3}{x^2-1}
\]
\[
= \frac{\sqrt{9} - 3}{0} = \frac{0}{0}
\]
