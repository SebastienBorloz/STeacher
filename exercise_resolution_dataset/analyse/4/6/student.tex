\[ \lim_{x \to +\infty} \frac{x(x+2) + x+2 - x^2}{x+2} \]
\[ = \lim_{x \to +\infty} \frac{x^2+2x+x+2-x^2}{x+2} \]
\text{a)}
\[ = \lim_{x \to +\infty} \frac{3x+2}{x+2} \]
\[ \lim_{x \to +\infty} \frac{x(3+2/x)}{x(1+2/x)}  \]
\[ = 3 \]

\[ \lim_{x \to 1} \frac{9 - \sqrt{16}}{\sqrt{4} - \sqrt{25}} \]
\[ = \frac{9 - 4}{2 - 5} = \frac{5}{-3} \]
\[ = -5/3 \]

\text{c)}
\[ \lim_{x \to 1} \frac{\sqrt{x^2+2x+6} - 3}{x^2 - 1} \]
\[ = \frac{6}{0} = 0 \]

\[ = \frac{x^2+2x-3}{(x^2-1)(\sqrt{x^2+2x+6}+3)} \]
\[ = \frac{(x+3)(x-1)}{(x+1)(x-1)(\sqrt{x^2+2x+6}+3)} \]
\[ \to \lim_{x \to 1} \frac{x+3}{(x+1)(\sqrt{x^2+2x+6}+3)} \]
\[ = \frac{4}{3(3+3)} \]
\[ = \frac{4}{18} = \frac{2}{9} \]
